# Instalación de Zabbix 7.0 en Kali (resumen de los pasos aplicados)
sudo apt install -y snmp zabbix-server-mysql zabbix-frontend-php zabbix-agent mariadb-server apache2
sudo systemctl start mariadb
# Base de datos y usuario
sudo mysql -e "CREATE DATABASE zabbix CHARACTER SET utf8mb4 COLLATE utf8mb4_bin;"
sudo mysql -e "CREATE USER 'zabbix'@'localhost' IDENTIFIED BY 'ClaveZabbixLab';"
sudo mysql -e "GRANT ALL PRIVILEGES ON zabbix.* TO 'zabbix'@'localhost';"
# Esquema (descargado de cdn.zabbix.com, zabbix-7.0.0.tar.gz)
cd /tmp/zabbix-7.0.0/database/mysql
sudo mysql zabbix < schema.sql
sudo mysql zabbix < images.sql
sudo mysql zabbix < data.sql
# Configuración del servidor y frontend
sudo sed -i 's/^# DBPassword=.*/DBPassword=ClaveZabbixLab/' /etc/zabbix/zabbix_server.conf
sudo a2enconf zabbix-frontend-php
sudo systemctl restart zabbix-server zabbix-agent apache2
sudo systemctl enable zabbix-server zabbix-agent apache2
